& {
$isAdmin = ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
if (-not $isAdmin) { Write-Host "Please run PowerShell as Administrator" -ForegroundColor Red; return }
$out = "$env:USERPROFILE\Desktop\lab-check2-$env:COMPUTERNAME.txt"
Start-Transcript -Path $out -Force | Out-Null
Write-Host "===== $env:COMPUTERNAME (round 2) =====" -ForegroundColor Cyan

Write-Host "`n[A] SQL instances, TCP ports, service accounts" -ForegroundColor Yellow
Get-CimInstance Win32_Service -Filter "Name LIKE 'MSSQL%'" | Select-Object Name, StartName, State | Format-Table -AutoSize | Out-Host
$inst = (Get-ItemProperty 'HKLM:\SOFTWARE\Microsoft\Microsoft SQL Server\Instance Names\SQL' -ErrorAction SilentlyContinue).PSObject.Properties | Where-Object { $_.Name -notlike 'PS*' }
foreach ($i in $inst) {
  $p = Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Microsoft SQL Server\$($i.Value)\MSSQLServer\SuperSocketNetLib\Tcp\IPAll" -ErrorAction SilentlyContinue
  Write-Host "Instance $($i.Name): TcpPort=$($p.TcpPort) DynamicPort=$($p.TcpDynamicPorts)"
}

Write-Host "`n[B] Databases, auth mode, logins (per instance)" -ForegroundColor Yellow
if (Get-Command sqlcmd -ErrorAction SilentlyContinue) {
  foreach ($i in $inst) {
    $s = if ($i.Name -eq 'MSSQLSERVER') { '.' } else { ".\$($i.Name)" }
    Write-Host "--- $s ---"
    sqlcmd -S $s -E -W -Q "SET NOCOUNT ON; SELECT name AS db FROM sys.databases; SELECT CASE SERVERPROPERTY('IsIntegratedSecurityOnly') WHEN 1 THEN 'WindowsOnly' ELSE 'Mixed' END AS auth_mode; SELECT name, type_desc, is_disabled FROM sys.server_principals WHERE type IN ('S','U','G') AND name NOT LIKE '##%' AND name NOT LIKE 'NT %';" | Out-Host
  }
} else { Write-Host "sqlcmd not found" }

Write-Host "`n[C] IIS feature, sites, web.config connection strings" -ForegroundColor Yellow
$appcmd = "$env:windir\system32\inetsrv\appcmd.exe"
if (Test-Path $appcmd) { & $appcmd list site | Out-Host } else { Write-Host "IIS not installed" }
if (Test-Path C:\inetpub\wwwroot) {
  Get-ChildItem C:\inetpub\wwwroot -Directory | Select-Object Name | Out-Host
  Get-ChildItem C:\inetpub\wwwroot -Recurse -Filter web.config -ErrorAction SilentlyContinue | ForEach-Object {
    Write-Host "--- $($_.FullName)"
    Select-String -Path $_.FullName -Pattern 'connectionString' | ForEach-Object { $_.Line.Trim() }
  }
}

Write-Host "`n[D] WampServer: ports, vhosts, site DB settings" -ForegroundColor Yellow
if (Test-Path C:\wamp64) {
  Get-ChildItem C:\wamp64\bin\apache\*\conf\httpd.conf | Select-String '^\s*Listen' | ForEach-Object { $_.Line.Trim() }
  Get-ChildItem C:\wamp64\bin\apache\*\conf\extra\httpd-vhosts.conf | Select-String 'ServerName|DocumentRoot|VirtualHost' | ForEach-Object { $_.Line.Trim() }
  Write-Host "--- C:\wamp64\www\CEH"
  Get-ChildItem C:\wamp64\www\CEH -ErrorAction SilentlyContinue | Select-Object -First 40 Name | Out-Host
  foreach ($f in 'C:\wamp64\www\DVWA\config\config.inc.php','C:\wamp64\www\CEH\wp-config.php') {
    if (Test-Path $f) { Write-Host "--- $f"; Select-String $f -Pattern "db_|DB_" | ForEach-Object { $_.Line.Trim() } }
  }
} else { Write-Host "WampServer not installed" }

Write-Host "`n[E] Active Directory users (DC only)" -ForegroundColor Yellow
if (Get-Command Get-ADUser -ErrorAction SilentlyContinue) {
  Get-ADUser -Filter * -Properties ServicePrincipalName, DoesNotRequirePreAuth, MemberOf, Description |
    Select-Object SamAccountName, Name, Enabled, DoesNotRequirePreAuth,
      @{n='SPN';e={$_.ServicePrincipalName -join '; '}},
      @{n='Groups';e={($_.MemberOf | ForEach-Object { ($_ -split ',')[0] -replace '^CN=' }) -join '; '}},
      Description | Format-List | Out-Host
  Get-ADComputer -Filter * | Select-Object Name, DNSHostName | Format-Table -AutoSize | Out-Host
} else { Write-Host "Not a DC (skipped)" }

Stop-Transcript | Out-Null
Write-Host "`nSaved to $out" -ForegroundColor Green
}
